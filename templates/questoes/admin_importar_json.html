Importar questões por JSON

Cole um JSON com várias questões para cadastrá-las no banco de questões individuais.

Formato esperado

Use materia como slug da matéria, conteudos como slugs dos conteúdos, conteudo_principal como slug e marque exatamente uma alternativa com correta: true. 

{
  "materia": "matematica",
  "questoes": [
    {
      "codigo": "ENEM-2024-MAT-CD7-140",
      "enunciado": "Texto da questão...",
      "explicacao": "Explicação da resolução...",
      "dificuldade": "medium",
      "tipo_fonte": "enem",
      "fonte_nome": "ENEM",
      "fonte_ano": 2024,
      "fonte_url": "https://...",
      "status": "draft",
      "conteudo_principal": "funcao-afim",
      "conteudos": ["funcao-afim", "matematica-basica"],
      "gabarito": "C",
      "requer_imagem": false,
      "alternativas": [
        {"chave": "A", "texto": "Alternativa A", "correta": false, "ordem": 1},
        {"chave": "B", "texto": "Alternativa B", "correta": false, "ordem": 2},
        {"chave": "C", "texto": "Alternativa C", "correta": true, "ordem": 3}
      ]
    }
  ]
}
